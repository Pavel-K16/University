# Примеры кода для синхронизации метрономов

## Вариант 1: Синхронизация В ФАЗЕ (рекомендуется для начала)

```go
func solveWithGraph(t0, T, dt float64) {
    graph := config.NewGraph()

    // Неподвижная платформа (основа)
    fixedPlatform := config.NewFixedNode(0, 0.0)
    graph.AddNode(fixedPlatform)

    // Метроном 1 - параметры из статьи Pena-Ramirez et al.
    metronome1 := config.NewMovableNode(1, 
        0.210,  // масса [кг] - из статьи
        0.001,  // начальное положение [м] - небольшое смещение
        0.0,    // начальная скорость [м/с]
        0.0,    // собственная жёсткость K
        0.0     // собственное демпфирование D
    )
    graph.AddNode(metronome1)

    // Подвижная платформа - ЛЁГКАЯ для синхронизации в фазе
    mobilePlatform := config.NewMovableNode(2,
        4.1,    // масса [кг] - ЛЁГКАЯ платформа (μ = 0.210/4.1 ≈ 0.051)
        0.0,    // начальное положение [м]
        0.0,    // начальная скорость [м/с]
        0.0,    // собственная жёсткость K
        0.0     // собственное демпфирование D
    )
    graph.AddNode(mobilePlatform)

    // Метроном 2 - идентичные параметры метроному 1
    metronome2 := config.NewMovableNode(3,
        0.210,  // масса [кг] - идентична метроному 1
        0.001,  // начальное положение [м] - близко к метроному 1 (для ускорения синхронизации)
        0.0,    // начальная скорость [м/с]
        0.0,    // собственная жёсткость K
        0.0     // собственное демпфирование D
    )
    graph.AddNode(metronome2)

    // Связи - параметры из статьи Pena-Ramirez et al.
    // Связь метроном 1 с платформой
    graph.AddEdge(1, 2,
        37.108,  // k [Н/м] - жёсткость пружины (из статьи)
        2.1378,  // d [Н·с/м] - демпфирование (из статьи)
        0.0      // rest - длина покоя
    )

    // Связь метроном 2 с платформой
    graph.AddEdge(3, 2,
        37.108,  // k [Н/м] - идентичная жёсткость
        2.1378,  // d [Н·с/м] - идентичное демпфирование
        0.0      // rest
    )

    // Связь платформы с неподвижной основой
    graph.AddEdge(2, 0,
        388.71,  // k [Н/м] - жёсткость связи с основой (из статьи)
        3.2656,  // d [Н·с/м] - демпфирование (из статьи)
        0.0      // rest
    )

    // Ожидаемое поведение:
    // - Синхронизация в фазе: x₁ ≈ x₃, ẋ₁ ≈ ẋ₃
    // - Платформа будет колебаться
    // - Частота ≈ собственная частота платформы (ω₃ ≈ 9.7 рад/с)
}
```

## Вариант 2: Синхронизация В ПРОТИВОФАЗЕ

```go
func solveWithGraph(t0, T, dt float64) {
    graph := config.NewGraph()

    fixedPlatform := config.NewFixedNode(0, 0.0)
    graph.AddNode(fixedPlatform)

    // Метроном 1
    metronome1 := config.NewMovableNode(1, 
        0.210,  // масса [кг]
        0.001,  // начальное положение [м]
        0.0,    // начальная скорость [м/с]
        0.0,    // собственная жёсткость K
        0.0     // собственное демпфирование D
    )
    graph.AddNode(metronome1)

    // Подвижная платформа - ТЯЖЁЛАЯ для синхронизации в противофазе
    mobilePlatform := config.NewMovableNode(2,
        8.818,  // масса [кг] - ТЯЖЁЛАЯ платформа (μ = 0.210/8.818 ≈ 0.024)
        0.0,    // начальное положение [м]
        0.0,    // начальная скорость [м/с]
        0.0,    // собственная жёсткость K
        0.0     // собственное демпфирование D
    )
    graph.AddNode(mobilePlatform)

    // Метроном 2 - противоположные начальные условия для ускорения синхронизации
    metronome2 := config.NewMovableNode(3,
        0.210,  // масса [кг]
        -0.001, // начальное положение [м] - противоположно метроному 1
        0.0,    // начальная скорость [м/с]
        0.0,    // собственная жёсткость K
        0.0     // собственное демпфирование D
    )
    graph.AddNode(metronome2)

    // Связи - те же параметры
    graph.AddEdge(1, 2, 37.108, 2.1378, 0.0)
    graph.AddEdge(3, 2, 37.108, 2.1378, 0.0)
    graph.AddEdge(2, 0, 388.71, 3.2656, 0.0)

    // Ожидаемое поведение:
    // - Синхронизация в противофазе: x₁ ≈ -x₃, ẋ₁ ≈ -ẋ₃
    // - Платформа стремится к покою (x₂ → 0)
    // - Частота ≈ собственная частота осцилляторов (ω ≈ 13.3 рад/с)
}
```

## Вариант 3: Исследование влияния массы платформы

```go
func solveWithGraph(t0, T, dt float64) {
    // Параметр для варьирования
    platformMass := 4.1  // Изменить на 8.818 для перехода к противофазе
    
    graph := config.NewGraph()

    fixedPlatform := config.NewFixedNode(0, 0.0)
    graph.AddNode(fixedPlatform)

    metronome1 := config.NewMovableNode(1, 0.210, 0.001, 0.0, 0.0, 0.0)
    graph.AddNode(metronome1)

    // Масса платформы варьируется
    mobilePlatform := config.NewMovableNode(2, platformMass, 0.0, 0.0, 0.0, 0.0)
    graph.AddNode(mobilePlatform)

    metronome2 := config.NewMovableNode(3, 0.210, 0.001, 0.0, 0.0, 0.0)
    graph.AddNode(metronome2)

    // Связи
    graph.AddEdge(1, 2, 37.108, 2.1378, 0.0)
    graph.AddEdge(3, 2, 37.108, 2.1378, 0.0)
    graph.AddEdge(2, 0, 388.71, 3.2656, 0.0)

    // Вычислить отношение масс для анализа
    mu := 0.210 / platformMass
    fmt.Printf("Отношение масс μ = %.4f\n", mu)
    if mu > 0.04 {
        fmt.Println("Ожидается синхронизация в фазе")
    } else {
        fmt.Println("Ожидается синхронизация в противофазе")
    }
}
```

## Ключевые изменения

1. **Массы метрономов:** уменьшить с 1.210 кг до **0.210 кг** (как в статье)
2. **Демпфирование:** увеличить с 0.1 до **2.1378 Н·с/м** (метроном-платформа) и **3.2656 Н·с/м** (платформа-основа)
3. **Масса платформы:** 
   - **4.1 кг** для синхронизации в фазе
   - **8.818 кг** для синхронизации в противофазе

## Ожидаемые результаты

### Синхронизация в фазе (m₂ = 4.1 кг):
- Разность фаз: Δ_phase → 0 рад
- Платформа: x₂ колеблется с постоянной амплитудой
- Частота: ≈ 1.5 Гц (частота платформы)

### Синхронизация в противофазе (m₂ = 8.818 кг):
- Разность фаз: Δ_phase → π рад
- Платформа: x₂ → 0 (стремится к покою)
- Частота: ≈ 2.1 Гц (частота осцилляторов)
